function hexToRgba(hex, alpha) {
  const h = hex.replace("#", "");
  const full = h.length === 3 ? h.split("").map((c) => c + c).join("") : h;
  const n = parseInt(full, 16);
  const r = (n >> 16) & 255,
    g = (n >> 8) & 255,
    b = n & 255;
  return `rgba(${r}, ${g}, ${b}, ${alpha})`;
}

async function askClaude(messages, system) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-6",
      max_tokens: 1000,
      system,
      messages,
    }),
  });
  const data = await res.json();
  const text = (data.content || [])
    .map((b) => (b.type === "text" ? b.text : ""))
    .join("\n")
    .trim();
  return text;
}

function parseQuestions(raw) {
  try {
    const cleaned = raw.replace(/```json|```/g, "").trim();
    const match = cleaned.match(/\[[\s\S]*\]/);
    const jsonStr = match ? match[0] : cleaned;
    const arr = JSON.parse(jsonStr);
    if (Array.isArray(arr) && arr.length > 0) {
      return arr.filter((q) => q && q.question && Array.isArray(q.options) && q.options.length >= 2);
    }
  } catch (e) {
    /* fall through to fallback */
  }
  return null;
}

function parseFlashcards(raw) {
  try {
    const cleaned = raw.replace(/```json|```/g, "").trim();
    const match = cleaned.match(/\[[\s\S]*\]/);
    const arr = JSON.parse(match ? match[0] : cleaned);
    if (Array.isArray(arr) && arr.length) {
      return arr.filter((c) => c && c.front && c.back);
    }
  } catch (e) {
    /* fall through */
  }
  return null;
}

const delay = (ms) => new Promise((r) => setTimeout(r, ms));

let voicesReadyPromise = null;
function ensureVoicesLoaded() {
  if (!("speechSynthesis" in window)) return Promise.resolve([]);
  if (voicesReadyPromise) return voicesReadyPromise;
  voicesReadyPromise = new Promise((resolve) => {
    const existing = window.speechSynthesis.getVoices();
    if (existing && existing.length) {
      resolve(existing);
      return;
    }
    let resolved = false;
    const finish = () => {
      if (resolved) return;
      resolved = true;
      resolve(window.speechSynthesis.getVoices());
    };
    window.speechSynthesis.onvoiceschanged = finish;
    setTimeout(finish, 700);
  });
  return voicesReadyPromise;
}

function speakText(text, onEnd) {
  if (!("speechSynthesis" in window)) return false;
  try {
    window.speechSynthesis.cancel();
    const clean = text.replace(/[#*_`]/g, "");
    ensureVoicesLoaded().then((voices) => {
      const utter = new SpeechSynthesisUtterance(clean);
      const fa = voices.find((v) => v.lang && v.lang.toLowerCase().startsWith("fa"));
      if (fa) utter.voice = fa;
      utter.lang = fa ? fa.lang : "fa-IR";
      utter.rate = 0.95;
      if (onEnd) utter.onend = onEnd;
      utter.onerror = () => onEnd && onEnd();
      window.speechSynthesis.speak(utter);
    });
    return true;
  } catch (e) {
    return false;
  }
}
function stopSpeaking() {
  if ("speechSynthesis" in window) window.speechSynthesis.cancel();
}

let pdfjsLoadPromise = null;
function loadPdfJs() {
  if (window.pdfjsLib) return Promise.resolve(window.pdfjsLib);
  if (pdfjsLoadPromise) return pdfjsLoadPromise;
  pdfjsLoadPromise = new Promise((resolve, reject) => {
    const timeoutId = setTimeout(() => {
      reject(new Error("pdfjs load timeout"));
    }, 8000);
    const script = document.createElement("script");
    script.src = "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js";
    script.onload = () => {
      clearTimeout(timeoutId);
      try {
        window.pdfjsLib.GlobalWorkerOptions.workerSrc =
          "https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js";
        resolve(window.pdfjsLib);
      } catch (e) {
        reject(e);
      }
    };
    script.onerror = () => {
      clearTimeout(timeoutId);
      reject(new Error("pdfjs load failed"));
    };
    document.head.appendChild(script);
  }).catch((e) => {
    pdfjsLoadPromise = null; // allow retry next time
    throw e;
  });
  return pdfjsLoadPromise;
}

async function extractPdfText(file, onPage) {
  const pdfjsLib = await loadPdfJs();
  const buf = await file.arrayBuffer();
  const pdf = await pdfjsLib.getDocument({ data: buf }).promise;
  const maxPages = Math.min(pdf.numPages, 40);
  let text = "";
  for (let i = 1; i <= maxPages; i++) {
    const page = await pdf.getPage(i);
    const content = await page.getTextContent();
    const pageText = content.items.map((it) => it.str).join(" ");
    text += pageText + "\n";
    if (onPage) onPage(i, maxPages);
  }
  return text;
}

function MasteryBar({ label, value, tone = "amber", delay: d = 0 }) {
  const [w, setW] = useState(0);
  useEffect(() => {
    const t = setTimeout(() => setW(value), 120 + d);
    return () => clearTimeout(t);
  }, [value, d]);
  const cvar = tone === "teal" ? "var(--accent2)" : tone === "coral" ? "var(--danger)" : "var(--accent)";
  return (
    <div className="mb-3">

        <div className="mark-fill" style={{ width: `${w}%`, background: `linear-gradient(90deg, ${cvar}, ${cvar})`, opacity: 0.9 }} />

function LiteMarkdown({ text }) {
  const lines = text.split("\n");
  return (
    <div>
      {lines.map((line, i) => {
        const t = line.trim();
        if (!t) return <div key={i} style={{ height: 8 }} />;
        if (t.startsWith("### ") || t.startsWith("## ") || t.startsWith("# ")) {
          return <p key={i} className="text-sm font-bold mt-3 mb-1" style={{ color: "var(--accent)" }}>{t.replace(/^#{1,3}\s*/, "")}</p>;
        }
        if (t.startsWith("- ") || t.startsWith("• ") || t.startsWith("* ")) {
          return <p key={i} className="text-xs leading-6" style={{ color: "var(--text)" }}>• {t.replace(/^[-*•]\s*/, "")}</p>;
        }
        return <p key={i} className="text-xs leading-6" style={{ color: "var(--muted)" }}>{t}</p>;
      })}
    </div>
  );
}

function Confetti() {
  const pieces = React.useMemo(
    () =>
      Array.from({ length: 22 }).map((_, i) => ({
        id: i,
        left: Math.random() * 100,
        delay: Math.random() * 0.4,
        duration: 1.6 + Math.random() * 1.2,
        rotate: Math.random() * 360,
        colorVar: ["var(--accent)", "var(--accent2)", "var(--text)"][i % 3],
        size: 6 + Math.random() * 6,
      })),
    []
  );
  return (
    <div className="confetti-wrap">
      {pieces.map((p) => (
        <span
          key={p.id}
          className="confetti-piece"
          style={{ left: `${p.left}%`, width: p.size, height: p.size * 0.4, background: p.colorVar, animationDelay: `${p.delay}s`, animationDuration: `${p.duration}s`, transform: `rotate(${p.rotate}deg)` }}
        />
      ))}
    </div>
  );
}

function InstallPromptView({ onContinue }) {
  const [deferredPrompt, setDeferredPrompt] = useState(null);
  const [installed, setInstalled] = useState(false);
  const [platform, setPlatform] = useState("other");

  useEffect(() => {
    const ua = typeof navigator !== "undefined" ? navigator.userAgent || "" : "";
    if (/iphone|ipad|ipod/i.test(ua)) setPlatform("ios");
    else if (/android/i.test(ua)) setPlatform("android");

    const handler = (e) => {
      e.preventDefault();
      setDeferredPrompt(e);
    };
    window.addEventListener("beforeinstallprompt", handler);

    const installedHandler = () => setInstalled(true);
    window.addEventListener("appinstalled", installedHandler);

    return () => {
      window.removeEventListener("beforeinstallprompt", handler);
      window.removeEventListener("appinstalled", installedHandler);
    };
  }, []);

  const handleInstallClick = async () => {
    if (!deferredPrompt) return;
    deferredPrompt.prompt();
    try {
      const choice = await deferredPrompt.userChoice;
      if (choice && choice.outcome === "accepted") setInstalled(true);
    } catch (e) {}
    setDeferredPrompt(null);
  };

  return (
    <div className="onboard fade-up">
      <div className="blob blob-a" />
      <div className="blob blob-b" />
      <div className="onboard-inner">
        <div className="onboard-icon pop-in">
          <Smartphone size={30} color="var(--contrast)" />
        </div>
        <h1 className="text-2xl font-black text-center mt-4" style={{ color: "var(--text)" }}>
          نصب اپ روی گوشیت
        </h1>
        <div className="flex justify-center">
          <Squiggle delay={200} />
        </div>
        <p className="text-xs text-center leading-6 mt-2 mb-6" style={{ color: "var(--muted)" }}>
          برنامه رو به صفحه اصلی گوشیت اضافه کن تا مثل یه اپ واقعی، سریع و تمام‌صفحه باز بشه.
        </p>

        {installed ? (
          <div className="onboard-field" style={{ animationDelay: "80ms" }}>
            <div className="flex items-center gap-2 justify-center" style={{ color: "var(--accent)" }}>
              <CheckCircle2 size={18} />
              <span className="text-sm font-bold">عالی، نصب شد! 🎉</span>
            </div>
          </div>
        ) : deferredPrompt ? (
          <div className="onboard-field" style={{ animationDelay: "80ms" }}>
            <button className="primary-btn" onClick={handleInstallClick}>
              <Smartphone size={16} />
              نصب برنامه
            </button>
          </div>
        ) : (
          <div className="onboard-field" style={{ animationDelay: "80ms" }}>
            {platform === "ios" ? (
              <div className="text-xs leading-7" style={{ color: "var(--text)" }}>
                <p className="font-bold mb-1">توی مرورگر سافاری:</p>

const STEPS = [
  { key: "uploading", label: "در حال آپلود" },
  { key: "extracting", label: "استخراج متن" },
  { key: "processing", label: "شناسایی ساختار" },
  { key: "ready", label: "آماده‌ست!" },
];

function UploadView({ book, onSetBook }) {
  const [status, setStatus] = useState("idle");
  const [errorMsg, setErrorMsg] = useState("");
  const [pageProgress, setPageProgress] = useState(null);
  const [pasteMode, setPasteMode] = useState(false);
  const [pasteTitle, setPasteTitle] = useState("");
  const [pasteText, setPasteText] = useState("");
  const fileRef = useRef(null);

  const runPipeline = async (title, rawText) => {
    setErrorMsg("");
    setStatus("uploading");
    await delay(350);
    setStatus("extracting");
    await delay(450);
    setStatus("processing");
    await delay(500);
    const trimmed = rawText.trim().slice(0, 9000);
    if (!trimmed) {
      setErrorMsg("متنی برای پردازش پیدا نشد.");
      setStatus("error");
      return;
    }
    onSetBook({ title: title || "کتاب من", chapter: "متن آپلودشده", source: "custom", text: trimmed });
    setStatus("ready");
  };

  const handleFile = async (file) => {
    const ext = file.name.split(".").pop().toLowerCase();
    const titleGuess = file.name.replace(/\.[^/.]+$/, "");
    try {
      if (ext === "txt" || ext === "md") {
        const text = await file.text();
        await runPipeline(titleGuess, text);
      } else if (ext === "docx") {
        setStatus("uploading");
        await delay(300);
        const buf = await file.arrayBuffer();
        setStatus("extracting");
        const result = await mammoth.extractRawText({ arrayBuffer: buf });
        await runPipeline(titleGuess, result.value || "");
      } else if (ext === "pdf") {
        setStatus("uploading");
        await delay(250);
        setStatus("extracting");
        let text;
        try {
          text = await extractPdfText(file, (i, total) => setPageProgress({ i, total }));
        } catch (pdfErr) {
          setPageProgress(null);
          setErrorMsg("این پیش‌نمایش نتونست فایل PDF رو بارگذاری کنه (معمولاً به‌خاطر محدودیت‌های امنیتی محیط پیش‌نمایش). فعلاً بهترین راه: فایل رو txt/docx کن یا متنش رو مستقیم پیست کن.");
          setStatus("error");
          return;
        }
        setPageProgress(null);
        if (!text.trim()) {
          setErrorMsg("متنی از این PDF استخراج نشد — احتمالاً یک نسخه اسکن‌شده (تصویری) است که این دمو نمی‌تواند OCR کند. فایل txt/docx آپلود کن یا متن رو پیست کن.");
          setStatus("error");
          return;
        }
        await runPipeline(titleGuess, text);
      } else {
        setErrorMsg("این فرمت پشتیبانی نمی‌شود. از pdf، docx یا txt استفاده کن.");
        setStatus("error");
      }
    } catch (e) {
      setErrorMsg("مشکلی در پردازش فایل پیش اومد. دوباره امتحان کن یا txt/docx آپلود کن.");
      setStatus("error");
    }
  };

  const onPick = (e) => {
    const file = e.target.files?.[0];
    if (file) handleFile(file);
  };
  const onDrop = (e) => {
    e.preventDefault();
    const file = e.dataTransfer.files?.[0];
    if (file) handleFile(file);
  };
  const submitPaste = () => {
    if (!pasteText.trim()) return;
    runPipeline(pasteTitle, pasteText);
  };
  const reset = () => {
    setStatus("idle");
    setErrorMsg("");
    setPasteMode(false);
    setPasteText("");
    setPasteTitle("");
    onSetBook(null);
  };

  const busy = ["uploading", "extracting", "processing"].includes(status);
  const stepIndex = STEPS.findIndex((s) => s.key === status);

  return (
    <div className="px-5 pt-6 pb-28 fade-up">
      <p className="text-xs" style={{ color: "var(--muted)" }}>کتاب من</p>
      <h2 className="text-2xl font-black mb-1" style={{ color: "var(--text)" }}>کتابتو آپلود کن</h2>
      <Squiggle delay={150} />
      <p className="text-xs leading-6 mt-3 mb-5" style={{ color: "var(--muted)" }}>
        فایل PDF، Word (docx) یا txt کتاب یا جزوه‌ت رو آپلود کن تا معلم خصوصی هوش مصنوعی از همون محتوا باهات چت کنه و سؤال بسازه. برای PDF فقط نسخه‌های متنی پشتیبانی می‌شن، نه اسکن‌شده — اگه PDF بالا نیومد، امن‌ترین راه فعلاً پیست‌کردن متنه.
      </p>

      {book && book.source === "custom" && !busy && (
        <div className="card mb-5" style={{ borderColor: "var(--accent2-border)" }}>
          <div className="flex items-start gap-3">
            <div className="ready-badge"><Check size={16} color="var(--contrast)" /></div>
            <div className="flex-1">
              <p className="text-sm font-bold" style={{ color: "var(--text)" }}>{book.title}</p>
              <p className="text-xs mt-1" style={{ color: "var(--muted)" }}>{book.text.length.toLocaleString("fa-IR")} نویسه — آماده برای چت و آزمون</p>
            </div>
            <button className="icon-btn" onClick={reset}><X size={15} color="var(--muted)" /></button>
          </div>
        </div>
      )}

      {busy && (
        <div className="card mb-5">
          <div className="flex flex-col gap-3">
            {STEPS.map((s, i) => (
              <div key={s.key} className="flex items-center gap-3">
                <div className="step-dot" style={{ background: i < stepIndex ? "var(--accent2)" : i === stepIndex ? "var(--accent)" : "var(--track)" }}>
                  {i < stepIndex && <Check size={11} color="var(--contrast)" />}
                  {i === stepIndex && <span className="spin-ring" />}
                </div>
                <span className="text-xs" style={{ color: i === stepIndex ? "var(--text)" : "var(--muted-strong)", fontWeight: i === stepIndex ? 700 : 500 }}>
                  {s.label}
                  {i === stepIndex && s.key === "extracting" && pageProgress ? ` — صفحه ${pageProgress.i} از ${pageProgress.total}` : ""}
                </span>
              </div>
            ))}
          </div>
        </div>
      )}

      {status === "error" && (
        <div className="card mb-5 flex items-start gap-2" style={{ borderColor: "var(--danger-border)" }}>
          <AlertCircle size={16} color="var(--danger)" className="shrink-0 mt-0.5" />
          <p className="text-xs leading-5" style={{ color: "var(--danger)" }}>{errorMsg}</p>
        </div>
      )}

      {!busy && !pasteMode && (
        <>
          <label className="dropzone" onDragOver={(e) => e.preventDefault()} onDrop={onDrop}>
            <UploadCloud size={26} color="var(--accent)" />
            <span className="text-sm font-bold mt-2" style={{ color: "var(--text)" }}>فایل رو بکش اینجا یا لمس کن</span>
            <span className="text-xs mt-1" style={{ color: "var(--muted-strong)" }}>pdf، docx، txt یا md</span>
            <input ref={fileRef} type="file" accept=".txt,.md,.docx,.pdf" className="hidden-input" onChange={onPick} />
          </label>
          <button className="btn-ghost w-full mt-3 flex items-center justify-center gap-2" onClick={() => setPasteMode(true)}>
            <FileText size={15} />
            یا متن رو مستقیم پیست کن
          </button>
        </>
      )}

      {!busy && pasteMode && (
        <div className="card">
          <input value={pasteTitle} onChange={(e) => setPasteTitle(e.target.value)} placeholder="عنوان کتاب یا درس..." className="paste-title" />
          <textarea value={pasteText} onChange={(e) => setPasteText(e.target.value)} placeholder="متن جزوه یا فصل کتابت رو اینجا بچسبون..." className="paste-area" rows={8} />
          <div className="flex gap-3 mt-3">
            <button className="btn-ghost flex-1" onClick={() => setPasteMode(false)}>انصراف</button>
            <button className="primary-btn flex-1" disabled={!pasteText.trim()} onClick={submitPaste}>پردازش کن</button>
          </div>
        </div>
      )}

      {!book && !busy && (
        <p className="text-xs text-center mt-6" style={{ color: "var(--muted-strong)" }}>
          تا وقتی آپلود نکنی، از یک کتاب نمونه درباره‌ی روش مطالعه استفاده می‌کنیم.
        </p>
      )}
    </div>
  );
}

function coachMessage(lastResult) {
  if (!lastResult) return "آماده‌ام کمکت کنم! از کجا شروع کنیم؟";
  const pct = Math.round((lastResult.score / lastResult.total) * 100);
  if (pct >= 70) return "دمت گرم! همینطوری ادامه بده 🎉";
  if (pct < 40) return "اشکالی نداره، بیا با هم دوباره تمرین کنیم.";
  return "خوب پیش رفتی، یکم دیگه تمرین کن تا عالی بشه.";
}

          <p className="text-xs leading-5" style={{ color: "var(--muted)" }}>بعد از اولین آزمونت روی این کتاب، نقاط ضعفت اینجا نمایش داده می‌شه.</p>
        </div>
      )}

      <div className="card">
        <h3 className="text-sm font-bold mb-3" style={{ color: "var(--text)" }}>آخرین آزمون</h3>
        {lastResult ? (
          <div className="flex items-center justify-between">
            <div>
              <p className="text-xs mb-1" style={{ color: "var(--muted)" }}>{lastResult.chapter}</p>
              <p className="text-lg font-black" style={{ color: "var(--accent2)" }}>{lastResult.score} از {lastResult.total}</p>
            </div>
            <button onClick={onGoExam} className="btn-ghost">آزمون جدید</button>
          </div>
        ) : (
          <button onClick={onGoExam} className="empty-cta">
            <ClipboardList size={18} color="var(--accent)" />
            <span>هنوز آزمونی نداده‌ای — یکی بساز</span>
          </button>
        )}
      </div>
    </div>
  );
}

function ChatView({ book, autoAsk, onConsumeAutoAsk, onUserMessage, onSaveNote, isOnline }) {
  const [messages, setMessages] = useState([
    { role: "assistant", content: `سلام! من معلم خصوصی توئم برای «${book.title}». هر سؤالی درباره این متن داری بپرس 🌱` },
  ]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const [speakingIdx, setSpeakingIdx] = useState(null);
  const [listening, setListening] = useState(false);
  const [savedIdx, setSavedIdx] = useState({});
  const recognitionRef = useRef(null);
  const scrollRef = useRef(null);
  const bookRef = useRef(book);
  bookRef.current = book;
  const didMountRef = useRef(false);

  useEffect(() => {
    if (!didMountRef.current) {
      didMountRef.current = true;
      return;
    }
    setMessages([{ role: "assistant", content: `سلام! من معلم خصوصی توئم برای «${book.title}». هر سؤالی درباره این متن داری بپرس 🌱` }]);
  }, [book.title]);

  useEffect(() => {
    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
  }, [messages, loading]);

  const send = useCallback(
    async (text) => {
      const q = (text ?? input).trim();
      if (!q || loading) return;
      const next = [...messages, { role: "user", content: q }];
      setMessages(next);
      setInput("");
      if (!isOnline) {
        setMessages((cur) => [...cur, { role: "assistant", content: "چت به اینترنت نیاز داره. وقتی دوباره وصل شدی برات جواب می‌دم — تا اون موقع می‌تونی از آزمون یا فلش‌کارت آفلاین استفاده کنی." }]);
        return;
      }
      setLoading(true);
      onUserMessage && onUserMessage();
      try {
        const b = bookRef.current;
        const system = `تو یک معلم خصوصی صمیمی و ساده‌گو برای یک دانش‌آموز دبیرستانی هستی. فقط و فقط بر اساس متن زیر از کتاب/جزوه دانش‌آموز پاسخ بده و از دانش عمومی خودت استفاده نکن مگر اینکه لازم باشد، که در آن صورت صریحاً بگو «این بخش در متن نبود». پاسخ کوتاه، ساده و مرحله‌به‌مرحله بده.\n\nعنوان: ${b.title}\nمتن:\n${b.text}`;
        const apiMessages = next.map((m) => ({ role: m.role, content: m.content }));
        const reply = await askClaude(apiMessages, system);
        setMessages((cur) => [...cur, { role: "assistant", content: reply || "متوجه نشدم، می‌شه دوباره بپرسی؟" }]);
      } catch (e) {
        setMessages((cur) => [...cur, { role: "assistant", content: "یه مشکلی پیش اومد، دوباره امتحان کن." }]);
      } finally {
        setLoading(false);
      }
    },
    [input, loading, messages, onUserMessage, isOnline]
  );

  useEffect(() => {
    if (autoAsk) {
      send(autoAsk);
      onConsumeAutoAsk && onConsumeAutoAsk();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const toggleSpeak = (i, text) => {
    if (speakingIdx === i) {
      stopSpeaking();
      setSpeakingIdx(null);
      return;
    }
    const ok = speakText(text, () => setSpeakingIdx(null));
    if (ok) setSpeakingIdx(i);
  };

  const toggleMic = () => {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SR) return;
    if (listening) {
      recognitionRef.current?.stop();
      setListening(false);
      return;
    }
    const rec = new SR();
    rec.lang = "fa-IR";
    rec.interimResults = true;
    rec.maxAlternatives = 1;
    rec.onresult = (e) => {
      let text = "";
      for (let i = 0; i < e.results.length; i++) text += e.results[i][0].transcript;
      setInput(text);
    };
    rec.onend = () => setListening(false);
    rec.onerror = () => setListening(false);
    recognitionRef.current = rec;
    rec.start();
    setListening(true);
  };

  return (
    <div className="flex flex-col h-full">
      <div className="px-5 pt-6 pb-3 fade-up">
        <p className="text-xs" style={{ color: "var(--muted)" }}>در حال چت درباره</p>
        <h2 className="text-lg font-black flex items-center gap-2" style={{ color: "var(--text)" }}>
          {book.chapter || book.title}
          <Sparkles size={16} color="var(--accent)" />
        </h2>
      </div>

      <div ref={scrollRef} className="flex-1 overflow-y-auto px-5 pb-3" style={{ scrollbarWidth: "none" }}>
        {messages.map((m, i) => (
          <div key={i} className={`bubble-row ${m.role === "user" ? "justify-start" : "justify-end"}`}>
            {m.role === "assistant" && (
              <div className="flex flex-col items-center gap-1.5">
                <button className="speak-btn" onClick={() => toggleSpeak(i, m.content)}>
                  {speakingIdx === i ? <Pause size={11} color="var(--accent)" /> : <Volume2 size={11} color="var(--muted)" />}
                </button>
                <button
                  className="speak-btn"
                  title="ذخیره در یادداشت‌ها"
                  onClick={() => { onSaveNote && onSaveNote(m.content); setSavedIdx((s) => ({ ...s, [i]: true })); }}
                >
                  {savedIdx[i] ? <Check size={11} color="var(--accent2)" /> : <StickyNote size={11} color="var(--muted)" />}
                </button>
              </div>
            )}
            <div className={m.role === "user" ? "bubble-user" : "bubble-ai"}>{m.content}</div>
          </div>
        ))}
        {loading && (
          <div className="bubble-row justify-end">
            <div className="bubble-ai"><TypingDots /></div>
          </div>
        )}
      </div>

      {messages.length < 2 && (
        <div className="px-5 pb-2 flex gap-2 overflow-x-auto" style={{ scrollbarWidth: "none" }}>
          {SUGGESTED_QUESTIONS.map((q) => (
            <button key={q} className="chip" onClick={() => send(q)}>{q}</button>
          ))}
        </div>
      )}

      <div className="px-5 pb-24 pt-2">
        <div className="input-bar">
          <button className="send-btn" disabled={loading || !input.trim()} onClick={() => send()}>
            <Send size={16} />
          </button>
          <input value={input} onChange={(e) => setInput(e.target.value)} onKeyDown={(e) => e.key === "Enter" && send()} placeholder={listening ? "در حال گوش دادن..." : "سؤالت رو بنویس..."} className="chat-input" />
          <button className={`mic-btn ${listening ? "mic-active" : ""}`} onClick={toggleMic} title="ورودی صوتی">
            {listening ? <MicOff size={15} /> : <Mic size={15} />}
          </button>
        </div>
      </div>
    </div>
  );
}

function FlashcardView({ book, onGenerated, isOnline }) {
  const [cards, setCards] = useState(null);
  const [loading, setLoading] = useState(false);
  const [idx, setIdx] = useState(0);
  const [flipped, setFlipped] = useState(false);
  const [recording, setRecording] = useState(false);
  const [recordings, setRecordings] = useState({});
  const [micError, setMicError] = useState(null);
  const [playingRecIdx, setPlayingRecIdx] = useState(null);
  const [mastery, setMastery] = useState({});
  const [weakOnly, setWeakOnly] = useState(false);
  const mediaRecorderRef = useRef(null);
  const chunksRef = useRef([]);
  const audioRef = useRef(null);

  const generate = async () => {
    setLoading(true);
    setFlipped(false);
    setMastery({});
    setWeakOnly(false);
    const withIds = (arr) => arr.map((c, i) => ({ ...c, id: i }));
    if (!isOnline) {
      setCards(withIds(FALLBACK_FLASHCARDS));
      setIdx(0);
      setLoading(false);
      onGenerated && onGenerated();
      return;
    }
    try {
      const system = `تو دستیار ساخت فلش‌کارت مطالعه هستی. فقط بر اساس متن زیر، ۶ تا ۸ فلش‌کارت به فارسی بساز. خروجی را فقط و فقط به صورت آرایه JSON معتبر برگردان، بدون توضیح اضافه، با این ساختار: [{"front":"سؤال یا مفهوم کوتاه","back":"پاسخ یا توضیح کوتاه"}]`;
      const user = `متن:\n${book.text}`;
      const raw = await askClaude([{ role: "user", content: user }], system);
      const parsed = parseFlashcards(raw);
      setCards(withIds(parsed && parsed.length ? parsed : FALLBACK_FLASHCARDS));
      setIdx(0);
      onGenerated && onGenerated();
    } catch (e) {
      setCards(withIds(FALLBACK_FLASHCARDS));
      setIdx(0);
    } finally {
      setLoading(false);
    }
  };

  const displayCards = (cards && weakOnly) ? cards.filter((c) => mastery[c.id] === "weak" || mastery[c.id] === "medium") : cards;

  const next = () => {
    setFlipped(false);
    setIdx((i) => Math.min(i + 1, (displayCards?.length || 1) - 1));
  };
  const prev = () => {
    setFlipped(false);
    setIdx((i) => Math.max(i - 1, 0));
  };

  const markMastery = (cardId, level) => {
    setMastery((m) => ({ ...m, [cardId]: level }));
  };

  const toggleRecord = async (cardId) => {
    if (recording) {
      mediaRecorderRef.current?.stop();
      setRecording(false);
      return;
    }
    setMicError(null);
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      const rec = new MediaRecorder(stream);
      chunksRef.current = [];
      rec.ondataavailable = (e) => chunksRef.current.push(e.data);
      rec.onstop = () => {
        const blob = new Blob(chunksRef.current, { type: "audio/webm" });
        const url = URL.createObjectURL(blob);
        setRecordings((r) => ({ ...r, [cardId]: url }));
        stream.getTracks().forEach((t) => t.stop());
      };
      mediaRecorderRef.current = rec;
      rec.start();
      setRecording(true);
    } catch (e) {
      setMicError("دسترسی به میکروفون ممکن نشد.");
    }
  };

  const playRecording = (cardId) => {
    const url = recordings[cardId];
    if (!url) return;
    if (!audioRef.current) audioRef.current = new Audio();
    audioRef.current.src = url;
    audioRef.current.onended = () => setPlayingRecIdx(null);
    audioRef.current.play();
    setPlayingRecIdx(cardId);
  };

  if (!cards && !loading) {
    return (
      <div className="px-5 pt-6 pb-28 fade-up flex flex-col items-center text-center">
        <p className="text-xs" style={{ color: "var(--muted)" }}>مرور سریع</p>
        <h2 className="text-2xl font-black mb-1" style={{ color: "var(--text)" }}>فلش‌کارت‌های {book.title}</h2>
        <Squiggle delay={150} />
        <p className="text-xs leading-6 mt-3 mb-6" style={{ color: "var(--muted)" }}>
          نکات کلیدی متن رو به‌صورت کارت‌های قابل‌ورق‌زدن بساز و سریع مرور کن.
        </p>
        <button className="primary-btn" onClick={generate}>
          <Sparkles size={16} />
          فلش‌کارت بساز
        </button>
      </div>
    );
  }

  if (loading) {
    return (
      <div className="px-5 pt-24 flex flex-col items-center text-center">
        <div className="pencil-spinner mb-5" />
        <p className="text-sm font-bold" style={{ color: "var(--text)" }}>داره فلش‌کارت می‌سازه...</p>
      </div>
    );
  }

  if (weakOnly && displayCards.length === 0) {
    return (
      <div className="px-5 pt-24 flex flex-col items-center text-center">
        <ThumbsUp size={32} color="var(--accent2)" />
        <p className="text-sm font-bold mt-3" style={{ color: "var(--text)" }}>همه کارت‌ها رو بلدی!</p>
        <button className="btn-ghost mt-4" onClick={() => setWeakOnly(false)}>نمایش همه کارت‌ها</button>
      </div>
    );
  }

  const safeIdx = Math.min(idx, displayCards.length - 1);
  const card = displayCards[safeIdx];
  return (
    <div className="px-5 pt-6 pb-28 fade-up flex flex-col items-center">
      <div className="flex items-center justify-between w-full mb-3">
        <p className="text-xs" style={{ color: "var(--muted)" }}>کارت {safeIdx + 1} از {displayCards.length}</p>
        <button className={`weak-filter-btn ${weakOnly ? "weak-filter-active" : ""}`} onClick={() => { setWeakOnly((w) => !w); setIdx(0); setFlipped(false); }}>
          فقط ضعیف‌ها
        </button>
      </div>

      <div className="flip-card" onClick={() => setFlipped((f) => !f)}>
        <div className={`flip-card-inner ${flipped ? "flipped" : ""}`}>
          <div className="flip-face flip-front">
            <span className="text-xs mb-2" style={{ color: "var(--muted)" }}>سؤال</span>
            <p className="text-base font-bold leading-7" style={{ color: "var(--text)" }}>{card.front}</p>
            <span className="text-[10px] mt-4" style={{ color: "var(--muted-strong)" }}>لمس کن برای دیدن جواب</span>
          </div>
          <div className="flip-face flip-back">
            <span className="text-xs mb-2" style={{ color: "var(--contrast)" }}>پاسخ</span>
            <p className="text-base font-bold leading-7" style={{ color: "var(--contrast)" }}>{card.back}</p>
          </div>
        </div>
      </div>

      <div className="flex gap-2 mt-4">
        <button className={`mastery-btn mastery-bad ${mastery[card.id] === "weak" ? "mastery-active" : ""}`} onClick={(e) => { e.stopPropagation(); markMastery(card.id, "weak"); }}>
          <ThumbsDown size={13} /> بلد نیستم
        </button>
        <button className={`mastery-btn mastery-mid ${mastery[card.id] === "medium" ? "mastery-active" : ""}`} onClick={(e) => { e.stopPropagation(); markMastery(card.id, "medium"); }}>
          <Meh size={13} /> نه‌چندان
        </button>
        <button className={`mastery-btn mastery-good ${mastery[card.id] === "good" ? "mastery-active" : ""}`} onClick={(e) => { e.stopPropagation(); markMastery(card.id, "good"); }}>
          <ThumbsUp size={13} /> بلدم
        </button>
      </div>

      <button className="speak-btn-lg mt-4" onClick={(e) => { e.stopPropagation(); speakText(flipped ? card.back : card.front); }}>
        <Volume2 size={14} color="var(--accent)" />
        <span className="text-xs font-bold" style={{ color: "var(--accent)" }}>خواندن با صدا</span>
      </button>

      <div className="flex items-center gap-3 mt-3">
        <button className={`speak-btn-lg ${recording ? "recording-active" : ""}`} onClick={(e) => { e.stopPropagation(); toggleRecord(card.id); }}>
          {recording ? <Square size={13} color="var(--danger)" /> : <Mic size={14} color="var(--accent2)" />}
          <span className="text-xs font-bold" style={{ color: recording ? "var(--danger)" : "var(--accent2)" }}>
            {recording ? "در حال ضبط... بزن تا تموم بشه" : "ضبط تلفظ خودت"}
          </span>
        </button>
        {recordings[card.id] && !recording && (
          <button className="speak-btn-lg" onClick={(e) => { e.stopPropagation(); playRecording(card.id); }}>
            {playingRecIdx === card.id ? <Pause size={13} color="var(--accent)" /> : <Play size={13} color="var(--accent)" />}
            <span className="text-xs font-bold" style={{ color: "var(--accent)" }}>پخش صدای خودت</span>
          </button>
        )}
      </div>
      {micError && <p className="text-[11px] mt-2" style={{ color: "var(--danger)" }}>{micError}</p>}

      <div className="flex gap-3 w-full mt-6">
        <button className="btn-ghost flex-1" onClick={prev} disabled={safeIdx === 0}>قبلی</button>
        <button className="btn-ghost flex-1" onClick={generate}>ساخت دوباره</button>
        <button className="btn-ghost flex-1" style={{ background: "var(--accent-soft2)", color: "var(--accent)" }} onClick={next} disabled={safeIdx === displayCards.length - 1}>بعدی</button>
      </div>
    </div>
  );
}

function ExamView({ onFinish, book, onAskAboutTopic, quickStart, onConsumeQuickStart, isOnline }) {
  const [step, setStep] = useState("setup");
  const [numQ, setNumQ] = useState(5);
  const [customNumQ, setCustomNumQ] = useState("");
  const [difficulty, setDifficulty] = useState("متوسط");
  const [questionType, setQuestionType] = useState("mc"); // mc | essay
  const [questions, setQuestions] = useState([]);
  const [current, setCurrent] = useState(0);
  const [answers, setAnswers] = useState([]);
  const [score, setScore] = useState(0);
  const [animScore, setAnimScore] = useState(0);
  const [isQuick, setIsQuick] = useState(false);
  const [essayDraft, setEssayDraft] = useState("");
  const [essayRevealed, setEssayRevealed] = useState(false);

  const start = async (opts) => {
    const n = opts?.numQ ?? numQ;
    const d = opts?.difficulty ?? difficulty;
    const qType = opts?.questionType ?? questionType;
    setStep("loading");
    if (!isOnline) {
      const qs = FALLBACK_QUESTIONS.slice(0, n);
      setQuestions(qs);
      setAnswers(new Array(qs.length).fill(null));
      setCurrent(0);
      setStep("taking");
      return;
    }
    try {
      const system = qType === "essay"
        ? `تو یک طراح سؤال آموزشی هستی. فقط بر اساس متن زیر سؤال تشریحی طراحی کن. خروجی را فقط و فقط به صورت یک آرایه JSON معتبر برگردان، بدون هیچ توضیح یا Markdown اضافه. هر آیتم دقیقاً این ساختار را داشته باشد:\n{"question": "...", "model_answer": "یک پاسخ نمونه کامل و دقیق", "topic": "..."}\nمقدار topic باید یک برچسب کوتاه ۲ تا ۴ کلمه‌ای فارسی برای مبحث سؤال باشد. همه سؤال‌ها تشریحی، به زبان فارسی، و در سطح سختی "${d}" باشند.`
        : `تو یک طراح سؤال آموزشی هستی. فقط بر اساس متن زیر سؤال طراحی کن. خروجی را فقط و فقط به صورت یک آرایه JSON معتبر برگردان، بدون هیچ توضیح یا Markdown اضافه. هر آیتم دقیقاً این ساختار را داشته باشد:\n{"question": "...", "options": ["...","...","...","..."], "correct_index": 0, "explanation": "...", "topic": "..."}\nمقدار topic باید یک برچسب کوتاه ۲ تا ۴ کلمه‌ای فارسی برای مبحثی باشد که سؤال درباره آن است. همه سؤال‌ها تستی با ۴ گزینه، به زبان فارسی، و در سطح سختی "${d}" باشند.`;
      const user = `متن:\n${book.text}\n\nتعداد سؤال: ${n}`;
      const raw = await askClaude([{ role: "user", content: user }], system);
      let qs;
      if (qType === "essay") {
        const parsed = parseQuestions(raw);
        qs = (parsed && parsed.length ? parsed : []).slice(0, n).map((q) => ({
          question: q.question,
          options: ["جواب درست دادم", "جواب اشتباه دادم"],
          correct_index: 0,
          explanation: q.model_answer || q.explanation || "",
          topic: q.topic,
          isEssay: true,
        }));
        if (!qs.length) qs = FALLBACK_QUESTIONS.slice(0, n);
      } else {
        const parsed = parseQuestions(raw);
        qs = (parsed && parsed.length ? parsed : FALLBACK_QUESTIONS).slice(0, n);
      }
      setQuestions(qs);
      setAnswers(new Array(qs.length).fill(null));
      setCurrent(0);
      setStep("taking");
    } catch (e) {
      const qs = FALLBACK_QUESTIONS.slice(0, n);
      setQuestions(qs);
      setAnswers(new Array(qs.length).fill(null));
      setStep("taking");
    }
  };

  useEffect(() => {
    if (quickStart) {
      setIsQuick(true);
      start({ numQ: 5, difficulty: "متوسط" });
      onConsumeQuickStart && onConsumeQuickStart();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const pick = (idx) => {
    const next = [...answers];
    next[current] = idx;
    setAnswers(next);
  };

  const finish = () => {
    const s = answers.reduce((acc, a, i) => acc + (a === questions[i].correct_index ? 1 : 0), 0);
    setScore(s);
    setStep("result");
    const mistakes = questions
      .map((q, i) => ({ q, i }))
      .filter(({ q, i }) => answers[i] !== q.correct_index)
      .map(({ q, i }) => ({
        topic: q.topic || q.question.slice(0, 24),
        question: q.question,
        yourAnswer: q.options[answers[i]] ?? "بی‌پاسخ",
        correctAnswer: q.options[q.correct_index],

        /* ignore */

  if (step === "setup") {

        <Squiggle delay={150} />

          <p className="text-sm font-bold mb-3" style={{ color: "var(--text)" }}>تعداد سؤال</p>

          <div className="flex gap-2">

      </div>
    );
  }

  if (step === "loading") {
    return (
      <div className="px-5 pt-24 flex flex-col items-center text-center">
        <div className="pencil-spinner mb-5" />
        <p className="text-sm font-bold" style={{ color: "var(--text)" }}>داره سؤال طراحی می‌کنه...</p>

      </div>
    );
  }

  if (step === "taking") {
    const q = questions[current];
    const total = questions.length;
    if (q.isEssay) {
      const marked = answers[current] != null;
      return (
        <div className="px-5 pt-6 pb-28 fade-up">
          <div className="flex items-center gap-2 mb-5">
            {questions.map((_, i) => (

            ))}
          </div>
          <p className="text-xs mb-2" style={{ color: "var(--muted)" }}>سؤال {current + 1} از {total} • تشریحی</p>
          <h3 className="text-lg font-bold mb-4 leading-8" style={{ color: "var(--text)" }}>{q.question}</h3>

          <textarea
            className="essay-textarea"
            rows={5}
            placeholder="پاسخت رو اینجا بنویس..."
            value={essayDraft}
            onChange={(e) => setEssayDraft(e.target.value)}
            disabled={essayRevealed}
          />

          {!essayRevealed ? (
            <button className="btn-ghost w-full mt-3" onClick={() => setEssayRevealed(true)}>
              نشون بده پاسخ نمونه
            </button>
          ) : (
            <>
              <div className="card mt-3" style={{ borderColor: "var(--accent-border)" }}>
                <p className="text-xs font-bold mb-1" style={{ color: "var(--accent)" }}>پاسخ نمونه</p>
                <p className="text-sm leading-6" style={{ color: "var(--text)" }}>{q.explanation}</p>
              </div>
              <p className="text-xs font-bold mt-4 mb-2" style={{ color: "var(--text)" }}>خودت رو با پاسخ نمونه مقایسه کن — چطور بود؟</p>
              <div className="flex gap-3">
                <button className={`option-card ${answers[current] === 0 ? "option-selected" : ""}`} onClick={() => pick(0)}>درست جواب دادم</button>
                <button className={`option-card ${answers[current] === 1 ? "option-selected" : ""}`} onClick={() => pick(1)}>اشتباه بود</button>
              </div>

          <div className="flex gap-3 mt-6">
            {current > 0 && <button className="btn-ghost flex-1" onClick={() => { setCurrent((c) => c - 1); setEssayDraft(""); setEssayRevealed(false); }}>قبلی</button>}
            {current < total - 1 ? (

              <button className="primary-btn flex-1" disabled={!marked} onClick={finish}>پایان آزمون</button>
            )}
          </div>
        </div>
      );

    return (
      <div className="px-5 pt-6 pb-28 fade-up">
        <div className="flex items-center gap-2 mb-5">
          {questions.map((_, i) => (

          ))}
        </div>
        <p className="text-xs mb-2" style={{ color: "var(--muted)" }}>سؤال {current + 1} از {total}</p>
        <h3 className="text-lg font-bold mb-5 leading-8" style={{ color: "var(--text)" }}>{q.question}</h3>
        <div className="flex flex-col gap-3 mb-6">

          ))}

          {current > 0 && <button className="btn-ghost flex-1" onClick={() => setCurrent((c) => c - 1)}>قبلی</button>}
          {current < total - 1 ? (
            <button className="primary-btn flex-1" disabled={answers[current] == null} onClick={() => setCurrent((c) => c + 1)}>بعدی</button>
          ) : (
            <button className="primary-btn flex-1" disabled={answers[current] == null} onClick={finish}>پایان آزمون</button>
          )}
        </div>
      </div>
    );
  }

  const weakTopics = (() => {
    const map = {};
    questions.forEach((q, i) => {
      if (answers[i] !== q.correct_index) {

      }
    });
    return Object.entries(map).map(([topic, count]) => ({ topic, count }));
  })();

  const pct = questions.length ? score / questions.length : 0;
  const goodScore = pct >= 0.7;
  const examMood = goodScore ? "happy" : pct < 0.4 ? "sad" : "idle";

          <RotateCcw size={16} />
          آزمون جدید
        </button>
      </div>
    </div>
  );
}

/* ------------------------------------------------------------------ */
/*  Pomodoro timer                                                      */
/* ------------------------------------------------------------------ */

function PomodoroView({ onBack, onSessionComplete }) {
  const FOCUS = 25 * 60;
  const BREAK = 5 * 60;
  const LONG_BREAK = 20 * 60;
  const [mode, setMode] = useState("focus"); // focus | break | longBreak
  const [secondsLeft, setSecondsLeft] = useState(FOCUS);
  const [running, setRunning] = useState(false);
  const [round, setRound] = useState(1);
  const [focusModeOn, setFocusModeOn] = useState(false);
  const [distractionCount, setDistractionCount] = useState(0);
  const [showDistractionToast, setShowDistractionToast] = useState(false);
  const intervalRef = useRef(null);

  useEffect(() => {
    if (!running) return;
    intervalRef.current = setInterval(() => {
      setSecondsLeft((s) => {
        if (s <= 1) {
          clearInterval(intervalRef.current);
          setRunning(false);
          if (mode === "focus") {
            onSessionComplete && onSessionComplete();
            const isLong = round % 4 === 0;
            setMode(isLong ? "longBreak" : "break");
            return isLong ? LONG_BREAK : BREAK;
          } else {
            setRound((r) => (mode === "break" || mode === "longBreak" ? r + 1 : r));
            setMode("focus");
            return FOCUS;
          }
        }
        return s - 1;
      });
    }, 1000);
    return () => clearInterval(intervalRef.current);
  }, [running, mode, round]);

  useEffect(() => {
    if (!focusModeOn || !running || mode !== "focus") return;
    const onVisibility = () => {
      if (document.hidden) {
        setDistractionCount((c) => c + 1);
        setShowDistractionToast(true);
      } else {
        setTimeout(() => setShowDistractionToast(false), 2500);
      }
    };
    document.addEventListener("visibilitychange", onVisibility);
    return () => document.removeEventListener("visibilitychange", onVisibility);
  }, [focusModeOn, running, mode]);

  const total = mode === "focus" ? FOCUS : mode === "break" ? BREAK : LONG_BREAK;
  const pct = 1 - secondsLeft / total;
  const R = 90;
  const C = 2 * Math.PI * R;
  const offset = C - pct * C;
  const mm = String(Math.floor(secondsLeft / 60)).padStart(2, "0");
  const ss = String(secondsLeft % 60).padStart(2, "0");

  const reset = () => {
    setRunning(false);
    setMode("focus");
    setSecondsLeft(FOCUS);
    setRound(1);
    setDistractionCount(0);
  };

  const skip = () => {
    setRunning(false);
    if (mode === "focus") {
      const isLong = round % 4 === 0;
      setMode(isLong ? "longBreak" : "break");
      setSecondsLeft(isLong ? LONG_BREAK : BREAK);
    } else {
      setRound((r) => r + 1);
      setMode("focus");
      setSecondsLeft(FOCUS);
    }
  };

  const label = mode === "focus" ? "زمان تمرکز" : mode === "break" ? "استراحت کوتاه" : "استراحت طولانی";

  return (
    <div className="px-5 pt-6 pb-28 fade-up flex flex-col items-center">
      <div className="flex items-center w-full mb-4">
        <button className="back-btn" onClick={onBack}><ChevronLeft size={18} color="var(--text)" /></button>
        <h2 className="text-lg font-black flex-1 text-center" style={{ color: "var(--text)" }}>پومودورو</h2>
        <div style={{ width: 32 }} />
      </div>

      <button
        className={`focus-toggle ${focusModeOn ? "focus-toggle-on" : ""}`}
        onClick={() => setFocusModeOn((f) => !f)}
      >
        <Focus size={14} />
        <span>حالت تمرکز {focusModeOn ? "فعال" : "غیرفعال"}</span>
      </button>

      {focusModeOn && running && mode === "focus" && (
        <p className="text-[11px] mt-2 text-center" style={{ color: "var(--muted)" }}>
          {showDistractionToast ? "برگشتی! بیا دوباره تمرکز کنیم 🙂" : distractionCount > 0 ? `${distractionCount} بار از این صفحه بیرون رفتی` : "از این تب بیرون نرو تا حواست جمع بمونه"}
        </p>
      )}

      <p className="text-xs mb-1 mt-1" style={{ color: "var(--muted)" }}>دور {round} • {label}</p>

      <div className="relative flex items-center justify-center my-4">
        <svg width="220" height="220" viewBox="0 0 220 220">
          <circle cx="110" cy="110" r={R} fill="none" stroke="var(--track)" strokeWidth="12" />
          <circle
            cx="110" cy="110" r={R} fill="none"
            stroke={mode === "focus" ? "var(--accent)" : "var(--accent2)"}
            strokeWidth="12" strokeLinecap="round" strokeDasharray={C} strokeDashoffset={offset}
            transform="rotate(-90 110 110)"
            style={{ transition: "stroke-dashoffset 1s linear" }}
          />
        </svg>
        <div className="absolute flex flex-col items-center">
          <span className="text-4xl font-black tabular-nums" style={{ color: "var(--text)", fontFamily: "monospace" }}>{mm}:{ss}</span>
          <span className="text-xs font-bold mt-1" style={{ color: "var(--muted)" }}>{mode === "focus" ? "تمرکز" : "استراحت"}</span>
        </div>
      </div>

      <div className="flex gap-3 w-full mt-4">
        <button className="btn-ghost flex-1" onClick={reset}><RotateCcw size={15} /> ریست</button>
        <button className="primary-btn flex-1" onClick={() => setRunning((r) => !r)}>
          {running ? <Pause size={16} /> : <Play size={16} />}
          {running ? "توقف" : "شروع"}
        </button>
        <button className="btn-ghost flex-1" onClick={skip}>رد کردن</button>
      </div>

      <p className="text-[11px] leading-5 mt-6 text-center" style={{ color: "var(--muted-strong)" }}>
        ۲۵ دقیقه تمرکز، ۵ دقیقه استراحت — بعد از ۴ دور یک استراحت طولانی. با هر دور تمرکز کامل، امتیاز می‌گیری.
      </p>
    </div>
  );
}

function reviewPlanFor(chapters) {
  return [...chapters]
    .sort((a, b) => a.mastery - b.mastery)
    .map((c) => {
      let when;
      if (c.mastery < 50) when = "امروز";
      else if (c.mastery < 75) when = "۲ روز دیگه";
      else when = "۵ روز دیگه";
      return { ...c, when };
    });
}

/* ------------------------------------------------------------------ */

function ProgressChartView({ onBack, history }) {
  const W = 320, H = 160, PAD = 20;
  const pts = history.map((h) => h.pct);
  const maxP = 100;
  const stepX = pts.length > 1 ? (W - PAD * 2) / (pts.length - 1) : 0;
  const coords = pts.map((p, i) => ({
    x: PAD + i * stepX,
    y: H - PAD - (p / maxP) * (H - PAD * 2),
  }));
  const path = coords.map((c, i) => `${i === 0 ? "M" : "L"}${c.x},${c.y}`).join(" ");
  const avg = pts.length ? Math.round(pts.reduce((a, b) => a + b, 0) / pts.length) : 0;

  return (
    <div className="px-5 pt-6 pb-28 fade-up">
      <div className="flex items-center w-full mb-5">
        <button className="back-btn" onClick={onBack}><ChevronLeft size={18} color="var(--text)" /></button>
        <h2 className="text-lg font-black flex-1 text-center" style={{ color: "var(--text)" }}>نمودار پیشرفت</h2>
        <div style={{ width: 32 }} />
      </div>

      {history.length === 0 ? (
        <div className="flex flex-col items-center text-center pt-12">
          <TrendingUp size={36} color="var(--accent2)" />
          <p className="text-sm font-bold mt-3" style={{ color: "var(--text)" }}>هنوز داده‌ای نیست</p>
          <p className="text-xs leading-5 mt-1" style={{ color: "var(--muted)" }}>بعد از چند آزمون، روند نمره‌هات رو اینجا می‌بینی.</p>
        </div>
      ) : (
        <>
          <div className="card mb-4">
            <div className="flex items-center justify-between mb-3">
              <h3 className="text-sm font-bold" style={{ color: "var(--text)" }}>روند درصد نمره‌ها</h3>
              <span className="text-xs font-bold" style={{ color: "var(--accent2)" }}>میانگین {avg}٪</span>
            </div>
            <svg width="100%" height={H} viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none">
              <line x1={PAD} y1={H - PAD} x2={W - PAD} y2={H - PAD} stroke="var(--track)" strokeWidth="1" />
              <path d={path} fill="none" stroke="var(--accent)" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
              {coords.map((c, i) => (
                <circle key={i} cx={c.x} cy={c.y} r="3.5" fill="var(--accent2)" />
              ))}
            </svg>
          </div>
          <div className="flex flex-col gap-2">
            {(() => {
              const lastSeen = {};
              const delta = {};
              history.forEach((h) => {
                if (lastSeen[h.chapter] !== undefined) {
                  delta[h.chapter] = h.pct - lastSeen[h.chapter];
                } else {
                  delta[h.chapter] = null;
                }
                lastSeen[h.chapter] = h.pct;
              });
              const rows = history.slice().reverse();
              const seen = new Set();
              return rows.map((h, i) => {
                const isLatestForChapter = !seen.has(h.chapter);
                seen.add(h.chapter);
                const d = isLatestForChapter ? delta[h.chapter] : null;
                return (
                  <div key={i}>
                    {d != null && d < 0 && (
                      <p className="regression-note">
                        نمره‌ی {h.chapter} نسبت به دفعه قبل کمتر شده — عیب نداره، یکم دیگه وقت بذاری درست می‌شه 💪
                      </p>
                    )}
                    {d != null && d > 0 && (
                      <p className="improvement-note">
                        نمره‌ی {h.chapter} نسبت به دفعه قبل {d}٪ بهتر شده — همینطوری ادامه بده 🎉
                      </p>
                    )}
                    <div className="flex items-center justify-between px-1">
                      <span className="text-xs" style={{ color: "var(--muted)" }}>{h.chapter}</span>
                      <span className="text-xs font-bold" style={{ color: "var(--text)" }}>{h.pct}٪</span>
                    </div>
                  </div>
                );
              });
            })()}
          </div>
        </>
      )}
    </div>
  );
}

function ReminderView({ onBack, enabled, onToggle, time, onTimeChange }) {
  const [permission, setPermission] = useState(

  };

  return (

      </div>

        {enabled ? <Bell size={20} color="var(--accent)" /> : <BellOff size={20} color="var(--muted)" />}
        <div className="flex-1">
          <p className="text-sm font-bold" style={{ color: "var(--text)" }}>یادآوری مطالعه روزانه</p>
          <p className="text-[11px] mt-0.5" style={{ color: "var(--muted)" }}>هر روز سر ساعت مشخص یادت می‌اندازیم بیای مطالعه کنی</p>
        </div>
        <button
          className={`toggle-switch ${enabled ? "toggle-on" : ""}`}
          onClick={() => (enabled ? onToggle(false) : requestPerm())}
        >
          <span className="toggle-knob" />
        </button>
      </div>

      {enabled && (
        <div className="card mb-4">
          <p className="text-sm font-bold mb-3" style={{ color: "var(--text)" }}>ساعت یادآوری</p>
          <input
            type="time"
            value={time}
            onChange={(e) => onTimeChange(e.target.value)}

            style={{ textAlign: "center" }}
          />
        </div>
      )}

      {permission === "denied" && (
        <p className="text-xs leading-5" style={{ color: "var(--danger)" }}>دسترسی نوتیفیکیشن مسدود شده — از تنظیمات مرورگر فعالش کن.</p>

      )}
    </div>
  );
}

function ParentReportView({ onBack, student, book, examsTaken, bestScorePct, xp, mistakes, pomodorosDone, scoreHistory }) {
  const [copied, setCopied] = useState(false);
  const recent = scoreHistory.slice(-5);
  const avgPct = recent.length ? Math.round(recent.reduce((a, r) => a + r.pct, 0) / recent.length) : null;
  const topWeak = (() => {
    const map = {};
    mistakes.forEach((m) => { map[m.topic] = (map[m.topic] || 0) + 1; });
    return Object.entries(map).sort((a, b) => b[1] - a[1]).slice(0, 3).map(([t]) => t);
  })();

  const reportText = `گزارش هفتگی مطالعه — ${student?.name || "دانش‌آموز"}
کتاب/موضوع: ${book.title}
تعداد آزمون‌های داده‌شده: ${examsTaken}
بهترین نمره: ${bestScorePct}٪
میانگین ۵ آزمون اخیر: ${avgPct != null ? avgPct + "٪" : "هنوز آزمونی نیست"}
جلسات پومودورو تکمیل‌شده: ${pomodorosDone}
امتیاز کل: ${xp} XP
${topWeak.length ? `مباحثی که نیاز به تمرین بیشتر دارن: ${topWeak.join("، ")}` : "نقطه ضعف خاصی ثبت نشده."}`;

  const copyReport = async () => {
    try {
      await navigator.clipboard.writeText(reportText);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch (e) {
      /* ignore */
    }
  };

  const shareReport = async () => {
    if (navigator.share) {
      try {
        await navigator.share({ title: "گزارش هفتگی مطالعه", text: reportText });
      } catch (e) {
        /* user cancelled or unsupported */
      }
    } else {
      copyReport();
    }
  };

  return (
    <div className="px-5 pt-6 pb-28 fade-up">
      <div className="flex items-center w-full mb-5">
        <button className="back-btn" onClick={onBack}><ChevronLeft size={18} color="var(--text)" /></button>
        <h2 className="text-lg font-black flex-1 text-center" style={{ color: "var(--text)" }}>گزارش هفتگی والدین</h2>
        <div style={{ width: 32 }} />
      </div>
      <div className="card mb-5">
        <pre className="text-xs leading-7 whitespace-pre-wrap" style={{ color: "var(--text)", fontFamily: "inherit" }}>{reportText}</pre>
      </div>
      <div className="flex gap-3">
        <button className="btn-ghost flex-1" onClick={copyReport}>
          <Copy size={14} />
          {copied ? "کپی شد!" : "کپی متن"}
        </button>
        <button className="primary-btn flex-1" onClick={shareReport}>
          <Share2 size={16} />
          اشتراک‌گذاری
        </button>
      </div>
    </div>
  );
}
